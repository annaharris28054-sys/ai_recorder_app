// ==========================================
// 1. Home / List View (搜索与列表)
// ==========================================
function HomeView({ onOpenRecord, onStartRecording }) {
  const [searchQuery, setSearchQuery] = useState('');

  // Search logic: matches title OR content (模拟搜索增强)
  const filteredRecords = MOCK_RECORDINGS.filter(r => 
    r.title.toLowerCase().includes(searchQuery.toLowerCase()) || 
    r.content.toLowerCase().includes(searchQuery.toLowerCase())
  );

  return (
    <div className="flex-1 flex flex-col h-full">
      <div className="px-6 pt-4 pb-2">
        <h1 className="text-2xl font-bold text-gray-900 mb-4">全部录音</h1>
        
        {/* Search Bar */}
        <div className="relative">
          <div className="absolute inset-y-0 left-0 pl-3 flex items-center pointer-events-none">
            <Search className="h-5 w-5 text-gray-400" />
          </div>
          <input
            type="text"
            className="block w-full pl-10 pr-3 py-3 border-none rounded-2xl bg-white shadow-sm focus:ring-2 focus:ring-indigo-500 text-sm placeholder-gray-400 transition-all"
            placeholder="搜索文件名或转写内容..."
            value={searchQuery}
            onChange={(e) => setSearchQuery(e.target.value)}
          />
        </div>
        
        {searchQuery && (
          <p className="text-xs text-indigo-500 mt-2 ml-1 flex items-center gap-1">
            <Sparkles size={12}/> 搜索增强已开启：正在检索AI转写全文
          </p>
        )}
      </div>

      {/* Record List */}
      <div className="flex-1 overflow-y-auto px-4 pb-4 mt-2">
        {filteredRecords.length > 0 ? (
          <div className="space-y-3">
            {filteredRecords.map((record) => (
              <div 
                key={record.id} 
                onClick={() => onOpenRecord(record)}
                className="bg-white p-4 rounded-2xl shadow-sm border border-gray-100 hover:border-indigo-100 cursor-pointer active:scale-[0.98] transition-all"
              >
                <div className="flex justify-between items-start mb-2">
                  <h3 className="font-semibold text-gray-800 truncate pr-4">{record.title}</h3>
                  <span className="text-xs text-gray-500 bg-gray-100 px-2 py-1 rounded-full shrink-0">{record.duration}</span>
                </div>
                <div className="flex items-center text-xs text-gray-400 mb-3 gap-3">
                  <span className="flex items-center gap-1"><Clock size={12} />{record.date}</span>
                  <span className="flex items-center gap-1 text-indigo-500 bg-indigo-50 px-1.5 rounded"><Sparkles size={12}/>已生成AI总结</span>
                </div>
                
                {/* 搜索时展示匹配上下文内容 */}
                {searchQuery && record.content.includes(searchQuery) && (
                  <div className="text-xs text-gray-600 bg-yellow-50 p-2 rounded-lg border border-yellow-100 line-clamp-2">
                    ... {record.content.substring(record.content.indexOf(searchQuery) - 10, record.content.indexOf(searchQuery) + 40)} ...
                  </div>
                )}
              </div>
            ))}
          </div>
        ) : (
          <div className="h-full flex flex-col items-center justify-center text-gray-400">
            <FileAudio size={48} className="mb-4 opacity-20" />
            <p className="text-sm">没有找到相关录音</p>
          </div>
        )}
      </div>
    </div>
  );
}

// ==========================================
// 2. Recording View (录音中页面)
// ==========================================
function RecordingView({ onStop }) {
  const [timer, setTimer] = useState(0);
  const [marks, setMarks] = useState([]);

  useEffect(() => {
    const interval = setInterval(() => setTimer(prev => prev + 1), 1000);
    return () => clearInterval(interval);
  }, []);

  const formatTime = (seconds) => {
    const m = Math.floor(seconds / 60).toString().padStart(2, '0');
    const s = (seconds % 60).toString().padStart(2, '0');
    return `${m}:${s}`;
  };

  const handleMark = () => {
    setMarks([...marks, timer]);
  };

  return (
    <div className="flex-1 flex flex-col bg-slate-900 text-white relative">
      <div className="px-6 pt-6 pb-4 flex justify-between items-center">
        <div className="flex items-center gap-2 bg-indigo-500/20 text-indigo-300 px-3 py-1.5 rounded-full text-xs font-medium">
          <Sparkles size={14} />
          AI 智能记录中...
        </div>
        <div className="text-xs text-slate-400 flex items-center gap-1">
           <VolumeX size={14}/> 深度降噪已开启
        </div>
      </div>

      {/* Center: Timer & Waveform */}
      <div className="flex-1 flex flex-col items-center justify-center -mt-10">
        <h2 className="text-6xl font-light font-mono mb-8 tracking-wider">{formatTime(timer)}</h2>
        
        {/* Animated Waveform Placeholder */}
        <div className="flex items-end gap-1.5 h-24 mb-12">
          {[...Array(15)].map((_, i) => (
            <div 
              key={i} 
              className="w-1.5 bg-indigo-400 rounded-full animate-pulse"
              style={{ 
                height: `${Math.max(20, Math.random() * 100)}%`,
                animationDelay: `${i * 0.1}s`,
                animationDuration: '0.8s'
              }}
            ></div>
          ))}
        </div>
        
        {/* Mark Indicator */}
        {marks.length > 0 && (
          <div className="text-slate-400 text-sm flex items-center gap-2">
            <Bookmark size={16} className="text-yellow-400" />
            已打点 {marks.length} 处关键节点
          </div>
        )}
      </div>

      {/* Real-time Transcription Preview */}
      <div className="h-40 bg-slate-800/50 rounded-t-3xl p-6 relative overflow-hidden">
        <div className="absolute top-0 left-0 w-full h-8 bg-gradient-to-b from-slate-800/50 to-transparent pointer-events-none"></div>
        <p className="text-slate-300 text-sm leading-relaxed opacity-80 animate-pulse">
          <span className="text-indigo-400 mr-2">[声纹1]</span>
          好的，我们现在开始会议。今天的主要议题是讨论第三季度的AI功能规划，大家有什么想法可以畅所欲言...
        </p>
      </div>

      {/* Controls */}
      <div className="bg-slate-800 pb-12 pt-6 px-8 flex justify-between items-center">
        <button 
          onClick={handleMark}
          className="w-14 h-14 rounded-full bg-slate-700 flex flex-col items-center justify-center gap-1 text-slate-300 active:bg-slate-600 transition-colors"
        >
          <Bookmark size={20} />
          <span className="text-[10px]">打点</span>
        </button>
        
        <button 
          onClick={onStop}
          className="w-20 h-20 rounded-full bg-red-500 flex items-center justify-center text-white shadow-lg shadow-red-500/30 active:scale-95 transition-transform"
        >
          <StopCircle size={40} />
        </button>

        <button className="w-14 h-14 rounded-full bg-slate-700 flex flex-col items-center justify-center gap-1 text-slate-300 active:bg-slate-600 transition-colors">
          <Pause size={20} />
          <span className="text-[10px]">暂停</span>
        </button>
      </div>
    </div>
  );
}

// 3. Details & Playback View (查看录音)
// ==========================================
function DetailsView({ record, onBack }) {
  const [activeTab, setActiveTab] = useState('summary'); // transcript, summary
  const [isPlaying, setIsPlaying] = useState(false);
  const [playbackSpeed, setPlaybackSpeed] = useState(1);

  const toggleSpeed = () => {
    if (playbackSpeed === 1) setPlaybackSpeed(1.5);
    else if (playbackSpeed === 1.5) setPlaybackSpeed(2);
    else setPlaybackSpeed(1);
  };

  return (
    <div className="flex-1 flex flex-col h-full bg-white relative">
      {/* Header */}
      <div className="px-4 py-3 flex items-center justify-between border-b border-gray-100 bg-white z-10 shrink-0">
        <button onClick={onBack} className="p-2 -ml-2 text-gray-600 hover:bg-gray-50 rounded-full">
          <ChevronLeft size={24} />
        </button>
        <div className="flex-1 px-4 truncate text-center font-semibold text-gray-800">
          {record.title}
        </div>
        <button className="p-2 text-gray-600 hover:bg-gray-50 rounded-full">
          <MoreVertical size={20} />
        </button>
      </div>

      {/* Tabs */}
      <div className="flex px-6 pt-4 gap-6 border-b border-gray-100 shrink-0">
        <button 
          onClick={() => setActiveTab('summary')}
          className={`pb-3 font-medium flex items-center gap-1.5 transition-colors relative ${activeTab === 'summary' ? 'text-indigo-600' : 'text-gray-400'}`}
        >
          <Sparkles size={16} />
          AI总结
          {activeTab === 'summary' && (
            <div className="absolute bottom-0 left-0 w-full h-0.5 bg-indigo-600 rounded-t-md"></div>
          )}
        </button>
        <button 
          onClick={() => setActiveTab('transcript')}
          className={`pb-3 font-medium flex items-center gap-1.5 transition-colors relative ${activeTab === 'transcript' ? 'text-indigo-600' : 'text-gray-400'}`}

          <FileText size={16} />

            <div className="absolute bottom-0 left-0 w-full h-0.5 bg-indigo-600 rounded-t-md"></div>
          )}
        </button>
      </div>

      {/* Content Scroll Area */}
      <div className="flex-1 overflow-y-auto bg-slate-50 p-5 pb-32">
        {activeTab === 'summary' ? (
          <div className="space-y-5 animate-in fade-in slide-in-from-bottom-2 duration-300">
            {/* Tags */}
            <div className="flex gap-2 flex-wrap">
              {record.summary.tags.map(tag => (

                <CheckCircle2 size={18} className="text-green-500" />
                核心结论
              </h3>
              <ul className="space-y-2">
                {record.summary.decisions.map((item, idx) => (
                  <li key={idx} className="text-sm text-gray-600 flex items-start gap-2">
                    <div className="w-1.5 h-1.5 rounded-full bg-gray-300 mt-1.5 shrink-0"></div>
                    {item}
                  </li>
                ))}
              </ul>
            </div>

            {/* ToDos */}
            <div className="bg-white p-4 rounded-2xl shadow-sm border border-gray-100">
              <h3 className="font-semibold text-gray-800 mb-3 flex items-center gap-2">
                <Clock size={18} className="text-orange-500" />

              </h3>
              <ul className="space-y-2">
                {record.summary.todos.map((item, idx) => (
                  <li key={idx} className="text-sm text-gray-600 flex items-start gap-2">
                    <div className="w-4 h-4 rounded border border-gray-300 shrink-0 mt-0.5"></div>
                    {item}
                  </li>
                ))}
              </ul>
            </div>
          </div>
        ) : (
          <div className="space-y-6 animate-in fade-in slide-in-from-bottom-2 duration-300">
            {record.transcript.map((line, idx) => (
              <div key={idx} className={`flex gap-3 ${line.isMarked ? 'bg-yellow-50/50 -mx-5 px-5 py-2' : ''}`}>
                <div className="w-12 shrink-0 pt-1 text-right">
                  <span className="text-xs text-gray-400 font-mono">{line.time}</span>
                </div>
                <div className="flex-1">
                  <div className="flex items-center gap-2 mb-1">
                    <span className="text-xs font-medium text-indigo-600 flex items-center gap-1">
                      <User size={12} /> {line.speaker}
                    </span>
                    {line.isMarked && (
                      <span className={`text-[10px] px-1.5 py-0.5 rounded ${line.markType === 'auto' ? 'bg-purple-100 text-purple-600' : 'bg-yellow-100 text-yellow-700'}`}>
                        {line.markType === 'auto' ? 'AI重点' : '手动打点'}
                      </span>
                    )}
                  </div>
                  <p className="text-sm text-gray-700 leading-relaxed">{line.text}</p>
                </div>
              </div>
            ))}
          </div>
        )}
      </div>

      {/* Audio Player Fixed Bottom */}
      <div className="absolute bottom-0 left-0 w-full bg-white border-t border-gray-100 p-5 rounded-t-3xl shadow-[0_-10px_40px_-15px_rgba(0,0,0,0.1)] z-20">
        {/* Progress Bar */}
        <div className="flex items-center gap-3 mb-4">
          <span className="text-[10px] text-gray-400 font-mono">12:05</span>
          <div className="flex-1 h-1.5 bg-gray-100 rounded-full relative overflow-hidden">

        </div>

        <div className="flex items-center justify-between px-2">
          {/* Speed Toggle */}

            onClick={toggleSpeed}
            className="w-10 text-xs font-semibold text-gray-500 hover:text-indigo-600 transition-colors"
          >
            {playbackSpeed.toFixed(1)}x
          </button>

          {/* Player Core Controls */}
          <div className="flex items-center gap-6">
            <button className="text-gray-400 hover:text-gray-700 flex flex-col items-center">
              <SkipBack size={24} />
              <span className="text-[8px] mt-1">退15s</span>
            </button>

            >

              <SkipForward size={24} />
              <span className="text-[8px] mt-1">进15s</span>

          </div>
